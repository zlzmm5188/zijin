    vip_level AS 'VIP等级',
    status AS '状态',
    parent_id AS '推荐人ID',

### 查询总账号下级用户数量
```sql
SELECT COUNT(*) AS '下级用户数'
FROM users
WHERE parent_id = 16;  -- 总账号ID
```

### 查询总账号下级用户列表
```sql
SELECT
    u.id,
    u.uid AS '邀请码',
    u.username AS '用户名',
    u.phone AS '手机号',
    u.vip_level AS 'VIP等级',
    u.total_invest AS '累计投资',
    u.created_at AS '注册时间'
FROM users u
WHERE u.parent_id = 16  -- 总账号ID
ORDER BY u.created_at DESC;
```

### 查询总账号下级用户充值记录
```sql
SELECT
    r.id,
    r.order_no AS '订单号',
    u.uid AS '用户邀请码',
    u.username AS '用户名',
    r.amount AS '充值金额',
    r.currency AS '币种',
    r.status AS '状态',
    r.created_at AS '充值时间'
FROM recharge_orders r
INNER JOIN users u ON r.user_id = u.id
WHERE u.parent_id = 16  -- 总账号ID
ORDER BY r.created_at DESC;
```

### 查询总账号下级用户充值总额
```sql

    r.currency AS '币种',
    SUM(r.amount) AS '充值总额',
    COUNT(*) AS '充值次数'

INNER JOIN users u ON r.user_id = u.id
WHERE u.parent_id = 16  -- 总账号ID
AND r.status = 2  -- 2=已完成

```
