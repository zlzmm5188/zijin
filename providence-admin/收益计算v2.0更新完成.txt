━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📝 核心变更

【删除概念】
❌ 年化收益率（annual_rate）
❌ 日利率（daily_rate）
❌ 总收益率（total_return_rate）

【新增概念】
✅ 周期天数（cycle_days）
✅ 基础收益率%（base_rate，整个周期）
✅ VIP额外收益率%（vip_extra_rate，整个周期）

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

💰 新收益计算公式

周期收益 = 投资金额 × (base_rate + vip_extra_rate) / 100
总金额 = 投资金额 + 周期收益
日均收益 = 周期收益 / cycle_days

⚠️ 注意：profit 与 rate 都是周期收益，不做年化换算

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📊 计算示例

【示例1】
投资金额: ¥10,000
周期: 30天
基础收益: 4.5%（30天）
VIP加成: +0.3%（VIP1）

计算:
周期收益 = 10000 × (4.5 + 0.3) / 100 = 480
总金额 = 10000 + 480 = 10,480
日均收益 = 480 / 30 = 16

【示例2】
投资金额: ¥50,000
周期: 60天
基础收益: 15%（60天）
VIP加成: +0.9%（VIP4）

计算:
周期收益 = 50000 × (15 + 0.9) / 100 = 7,950
总金额 = 50000 + 7950 = 57,950
日均收益 = 7950 / 60 = 132.5
