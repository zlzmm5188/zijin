#### API请求优化
```javascript
// 1. 请求去重（避免重复请求）
const requestCache = new Map();
async function cachedFetch(url, ttl = 30000) {
    const cached = requestCache.get(url);
    if (cached && Date.now() - cached.time < ttl) {
        return cached.data;
    }
    
    const data = await fetch(url).then(r => r.json());
    requestCache.set(url, { data, time: Date.now() });
    return data;
}

// 2. 请求合并（批量接口）
async function batchRequest(endpoints) {
    return await fetch('/api/batch', {
        method: 'POST',
        body: JSON.stringify({ requests: endpoints })
    });
}

// 3. 分页加载（无限滚动）
async function loadMoreRecords() {
    const records = await fetch(`/api/records?page=${page}`);
    // 追加到现有列表
}
```
